            <h1>AppleScript make command</h1>


26
            <p>


27
                The <code>make</code> command creates scriptable objects. SwiftTag uses it


28
                for creating track <a href="applescript-tag.html"><code>tag</code></a> and


29
                <a href="applescript-picture.html"><code>picture</code></a> objects.

                <h2 id="examples">Examples</h2>


130
                <div class="code-label">AppleScript</div>


131
                <pre><code>tell application "SwiftTag"


132
    tell track 1 of front editor window


133
        -- create a new custom tag --


134
        set newTag to make new tag with properties {key:"TEST", value:"This is a test"}


135
        


136
        -- create a tag that matches a defined track property --


137
        make new tag with properties {key:"COMMENT", value:"Reviewed"}


138
        -- which is the same as --


139
        set comment to "Reviewed"


140
    end tell


141
end tell</code></pre>


142
                <div class="code-label">AppleScript</div>


143
                <pre><code>tell application "SwiftTag"


144
    set newWindow to make new editor window


145
    tell newWindow


146
        set newWindowBounds to bounds


147
        set bounds to {x: (x of newWindowBounds) + 100, y: (y of newWindowBounds), width: 1100, height: 900}


148
    end tell


149
end tell</code></pre>
