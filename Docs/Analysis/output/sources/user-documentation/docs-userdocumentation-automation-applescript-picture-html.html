            <h1>AppleScript picture class</h1>


26

27

28
                <a href="applescript-track.html"><code>track</code></a>. It exposes


29
                picture type, description, image data, derived dimensions, and SwiftTag


30

                <h2 id="access">Access</h2>


67
                <p>


68
                    Pictures are read-write elements of the <a href="applescript-track.html"><code>track</code></a>


69
                    class. Use <a href="applescript-make.html"><code>make</code></a> to create


70
                    a picture and <a href="applescript-delete.html"><code>delete</code></a> to remove one.


71
                </p>


72
                <div class="code-label">AppleScript</div>


73
                <pre><code>tell application "SwiftTag"


74
    tell track 1 of front editor window


75
        set coverPicture to first picture whose picture type is front cover


76
        set coverDescription to description of coverPicture


77
    end tell


78
end tell</code></pre>

                <h2 id="examples">Examples</h2>


201
                <div class="code-label">AppleScript</div>


202
                <pre><code>tell application "SwiftTag"


203
    tell track 1 of front editor window


204
        set coverPicture to first picture whose picture type is front cover


205
        set coverSize to {width of coverPicture, height of coverPicture}


206
        set description of coverPicture to "Front cover"


207
    end tell


208
end tell</code></pre>


209
                <div class="code-label">AppleScript</div>


210
                <pre><code>use framework "Foundation"


211
use scripting additions


212


213
tell application "SwiftTag"


214
     tell track 1 of front editor window


215
        -- create picture from memory (binary/base64-encoded image data) --


216
        make new picture with properties {data:imageData, picture type:performance, description:"center stage close-up"}


217


218
        -- move a picture from one FLAC picture type to another --


219
        set leafletPicture to first picture whose picture type is leaflet


220
        set leafletPictureData to data of leafletPicture


221
        make new picture with data leafletPictureData with properties {picture type:media, description:"Leaflet/Media"}


222
        log "Moved picture with pool id " & pool id of result & " from leaflet to media with description " & quote & description of result & quote & "."


223
        delete leafletPicture


224


225
        -- create picture from file using NSData and Base64 encoded data --


226
        set pictureFilePath to "/path/to/image.jpg"


227
        set newLeafletPictureData to my getBase64EncodedData from pictureFilePath


228
        set newLeafletPicture to make new picture with properties {data:newLeafletPictureData, picture type:leaflet, description:"Leaflet"}


229


230
        -- create picture from file using AppleScript data --


231
        make new picture with properties {data:(my getData from "/path/to/another-image.png"), picture type:back cover, description:"Back Cover"}


232
    end tell


233
end tell


234


235
-- Base64 encode file data for AppleScript/NSData friendly binary data handling --


236
on getBase64EncodedData from filePath


237
    try


238
        set fileURL to current application's |NSURL|'s fileURLWithPath:filePath


239
        set fileData to current application's NSData's dataWithContentsOfURL:fileURL


240
        if fileData is missing value then


241
            error "ERROR: Could not read data from file. Check the path."


242
        end if


243
        return (fileData's base64EncodedStringWithOptions:0) as text


244
    on error errorMessage number errorNumber


245
        error "ERROR: Failed to get byte data. " & errorMessage & " (" & errorNumber & ")"


246
    end try


247
end getBase64EncodedData


248


249
-- Get file data as AppleScript data for direct assignment to picture data property (not base64 encoded) --


250
on getData from filePath


251
    try


252
        set fileAlias to POSIX file filePath as alias


253
    on error


254
         error "ERROR: Could not get file alias. Check the path."


255
    end try


256


257
    set fileHandle to open for access fileAlias


258
    try


259
        set fileData to read fileAlias to eof as data


260
        close access fileHandle


261
        return fileData


262
    on error errMsg number errNum


263
        try


264
            close access fileHandle


265
        end try


266
        error errMsg number errNum


267
    end try


268
end getData</code></pre>


269
                <aside class="callout note">


270
                    The <code>getData</code> handler in the above example does not require <code>use framework "Foundation"</code> 


271
                    or <code>use scripting additions</code>.


272
                </aside>


273
                <aside class="callout note">


274
                    SwiftTag will accept any macOS supported image type and will convert images that are not JPEG or PNG to PNG upon 


275
                    saving to FLAC file.


276
                </aside>
