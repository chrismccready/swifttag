            <h2 id="steps-apple-script">Multi-track edit via AppleScript</h2>


34
                <div class="code-label">AppleScript</div>


35
                <pre><code>set parentFolder to POSIX path of ((path to me as text) & "::")


36
set testFLACPath_1 to parentFolder & "path_to_your_flac/test_1.flac"


37
set testFLACPath_2 to parentFolder & "path_to_your_flac/test_2.flac"


38


39
tell application "SwiftTag"


40
    -- add tracks to editor window --


41
    set frontWindow to front editor window


42
    set addedTracks to (add POSIX file testFLACPath_1 to frontWindow) as list


43
    set end of addedTracks to (add POSIX file testFLACPath_2 to frontWindow)


44


45
    -- edit tags on all added tracks --


46
    repeat with thisTrack in addedTracks


47
        tell thisTrack


48
            set album to "Test Album"


49
            set album artist to "Test Artist"


50
            set sort album artist to "Artist, Test"


51
            set artist to "Test Artist"


52
            set sort artist to "Artist, Test"


53
        end tell


54
    end repeat


55


56
    -- save only tag changes to all and only added tracks --


57
    set selected tracks of frontWindow to addedTracks


58
    save frontWindow with scope selected with payload tags only


59
end tell</code></pre>
